transport.on('refusal', ({ address, authority, origin, port, reason, status }) => {
  // une page d'un autre site, par rebinding DNS, écrit :
  // refused 403 host-refused: authority "rebind.example:8080",
  //   origin "http://rebind.example:8080", from 127.0.0.1:53114
  console.warn(
    `refusé ${status} ${reason} : autorité ${JSON.stringify(authority)}, ` +
      `origine ${JSON.stringify(origin)}, de ${address}:${port}`
  );
});
