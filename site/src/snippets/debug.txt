const filter = () => {
  return process.env['DEBUG_CALLS'];
};

services.on('debug', filterDebug((error, info) => {
  console.log(info.direction, info.procedureName, error);
}, filter));
router.on('debug', filterEnvelopeDebug(({ direction, header }) => {
  console.log(direction, header);
}, filter));
